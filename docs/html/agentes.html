590    def propor_plano(self):
591        """Cria um plano de descarga (caudal, atraso, duração).
592        O plano é um dicionário. Quanto mais cheio o reservatório, maior o
593        caudal proposto (nunca acima do máximo seguro).
594
595        Returns:
596            dict | None: O plano, ou None se não é preciso descarregar.
597        """
598        if not self.avaliar_descarga():
599            return None
600
601        # fração do "espaço de risco" já ocupada: 0.0 no nível de alerta, 1.0 a 100%
602        excesso = (self.nivel_reservatorio - NIVEL_ALERTA_RESERVATORIO) / (
603            100.0 - NIVEL_ALERTA_RESERVATORIO)
604        excesso = min(excesso, 1.0)
605
606        return {
607            "caudal": round(self.caudal_maximo_seguro * excesso, 1),
608            "atraso": 0,                  # passos de simulação até começar
609            "duracao": DURACAO_DESCARGA,  # passos de simulação a descarregar
610        }

643    def validar_execucao(self, plano):
644        """Valida um plano antes de abrir as comportas, com o estado atual.
645
646        Vale para planos votados e para a contingência: nenhum voto autoriza
647        uma operação insegura (secção 5.2 do guia).
648
649        Args:
650            plano (dict): Caudal, atraso e duração.
651
652        Returns:
653            tuple: (seguro, razao). `razao` explica a recusa e é "" se seguro.
654        """
655        if not self.validar_plano(plano):
656            return False, "fora dos limites de segurança"
657        if self.ambiente.descarga_ativa():
658            return False, "já há uma descarga em curso"
659        if self.nivel_reservatorio <= NIVEL_ALERTA_RESERVATORIO:
660            return False, "o reservatório já desceu do nível de alerta"
661        return True, ""

663    def validar_plano(self, plano):
664        """Verifica se um plano respeita os limites de segurança.
665
666        Só olha para o plano. O estado do reservatório conta em `validar_execucao`.
667
668        Args:
669            plano (dict | None): Caudal, atraso e duração.
670
671        Returns:
672            bool: True se o caudal está entre 0 (exclusivo) e o máximo seguro e
673            o atraso e a duração são válidos. None não é um plano válido.
674        """
675        if plano is None:
676            return False
677        return (0 < plano["caudal"] <= self.caudal_maximo_seguro
678                and plano["atraso"] >= 0
679                and plano["duracao"] > 0)

685
686        Args:
687            plano (dict): Caudal, atraso e duração, já validados.
688        """
689        self.ambiente.iniciar_descarga(plano["caudal"], plano["duracao"],
690                                       plano["atraso"])

168    class OuvirMensagens(CyclicBehaviour):
169        """Recebe alertas do operador do bairro, as respostas dele aos pedidos
170        de mitigação e as propostas da barragem, e transforma-os em crenças
171        do agente BDI."""
172
173        async def run(self):
174            """Espera uma mensagem e converte-a numa crença do `lider.asl`
175            (`risco`, `falta`, `proposta`, `votacao` ou `contingencia`), que
176            dispara os planos de decisão."""
177            lider = self.agent
178            msg = mensagens.ler(await self.receive(timeout=1))
179            if msg is None:
180                return
181
182            if msg.tipo == "alerta_risco":
183                lider.risco_do_bairro = msg.corpo["risco"]
184                lider.bdi.set_belief("risco", atomo(msg.corpo["risco"]))
185
186            elif msg.protocolo == mensagens.PEDIR_MITIGACAO:
187                # o operador diz o que não tinha em estoque (AGREE e INFORM
188                # repetem a mesma falta, por isso só conta a primeira resposta)
189                if msg.tipo in TIPOS_COM_FALTA:
190                    lider.registar_falta(msg.corpo.get("falta") or {})
191
192            elif msg.protocolo == mensagens.GERIR_OFERTA:
193                # resposta do diretório à publicação das ofertas. O diretório já
194                # mostra o evento OFERTA; aqui só se avisa se as recusou
195                if msg.performativa == mensagens.REFUSE:
196                    registo.evento("OFERTA", f"bairro {lider.bairro}: o diretório "
197                                             f"recusou as ofertas "
198                                             f"({msg.corpo.get('motivo')})")
199                else:
200                    registo.detalhe(lider.nome, "o diretório registou as ofertas")
201
202            elif msg.tipo == "proposta_descarga":
203                if not lider.rede.pode_comunicar(lider.nome, "barragem"):
204                    return
205                lider._seq_propostas += 1
206                id_proposta = lider._seq_propostas
207                lider.propostas[id_proposta] = msg      # guarda o thread para responder
208                lider.bdi.set_belief("proposta", id_proposta,
209                                     float(msg.corpo["caudal"]))
210
211            elif msg.tipo == "pedido_voto":
212                if not lider.rede.pode_comunicar(lider.nome, "barragem"):
213                    return
214                lider._seq_propostas += 1
215                id_voto = lider._seq_propostas
216                lider.votacoes[id_voto] = msg           # guarda o thread para votar
217                lider.bdi.set_belief("votacao", id_voto, float(msg.corpo["caudal"]))
218
219            elif msg.tipo == "aviso_contingencia":
220                # a barragem decidiu sozinha: o líder fica a saber (crença) e o
221                # plano do .asl regista-o
222                lider.bdi.set_belief("contingencia", float(msg.corpo["caudal"]))

348        """Responde à barragem (ação `.responder`).

352            decisao (str): "aceita", "contraproposta" ou "rejeita".

367        self.submit(self.bdi.send(mensagens.criar(

370            mensagens.NEGOCIAR_DESCARGA, "resposta_proposta",
371            thread=proposta.thread, id_proposta=proposta.corpo.get("id_proposta"),
372            rodada=proposta.corpo.get("rodada"),
373            aceita=aceita, contraproposta=contraproposta)))
374        registo.detalhe(self.nome, f"{'aceitou' if aceita else 'recusou'} o plano "
375                                   f"(risco {self.risco_do_bairro}); contraproposta "
376                                   f"{contraproposta}")
