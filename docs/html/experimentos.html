experimentos

Experimentos: corre cenários mudando um parâmetro de cada vez (tarefa t27).

Cada experimento é uma execução normal de main.py <cenário>, num processo à parte, com duas variáveis de ambiente:

  • FLOOD_PARAMETROS (JSON) muda constantes do config.py, por exemplo {"MAX_RODADAS": 1} (ver o fim do config.py);
  • FLOOD_SAIDA guarda as métricas em resultados/experimentos/, para não sobrescrever os resultados que vão para os slides.

No fim junta o resumo de cada execução numa tabela: resultados/experimentos/comparacao.csv. A tabela acumula: voltar a correr um experimento substitui a sua linha.

Correr com um Python que tenha o SPADE (cada execução demora cerca de 1,5 minutos):

python experimentos.py --lista          mostra o plano, sem correr nada
python experimentos.py                  corre todos os grupos
python experimentos.py base rodadas     corre só estes grupos

Os grupos mudam um parâmetro de cada vez e comparam-se com a linha "base" do mesmo cenário (corra primeiro o grupo "base").

 59def pasta_do(nome, cenario):
 60    """Pasta onde um experimento guarda as suas métricas.
 61
 62    Args:
 63        nome (str): Nome do experimento, como "1 rodada".
 64        cenario (str): Uma chave de `CENARIOS`.
 65
 66    Returns:
 67        str: O caminho da pasta, com o nome limpo de caracteres especiais.
 68    """
 69    seguro = "".join(c if c.isalnum() else "_" for c in nome)
 70    return os.path.join(PASTA, seguro, cenario)

 73def correr(nome, cenario, parametros, tempo_maximo=300):
 74    """Corre `main.py <cenário>` com os parâmetros dados.
 75
 76    Args:
 77        nome (str): Nome do experimento, como "1 rodada".
 78        cenario (str): Uma chave de `CENARIOS`.
 79        parametros (dict): Constantes do config.py a mudar.
 80        tempo_maximo (int): Segundos antes de desistir da execução.
 81
 82    Returns:
 83        dict: O `resumo.json` que a execução gravou.
 84
 85    Raises:
 86        RuntimeError: Se a execução falhar ou não gravar o resumo.
 87    """
 88    saida = pasta_do(nome, cenario)
 89    ambiente = dict(os.environ, FLOOD_PARAMETROS=json.dumps(parametros),
 90                    FLOOD_SAIDA=saida, PYTHONIOENCODING="utf-8")
 91    processo = subprocess.run([sys.executable, "main.py", cenario], cwd=RAIZ,
 92                              env=ambiente, capture_output=True, text=True,
 93                              encoding="utf-8", errors="replace", timeout=tempo_maximo)
 94    ficheiro = os.path.join(saida, "resumo.json")
 95    if processo.returncode != 0 or not os.path.isfile(ficheiro):
 96        fim = "\n".join(processo.stderr.strip().splitlines()[-5:])
 97        raise RuntimeError(f"{nome} em {cenario} falhou (código "
 98                           f"{processo.returncode}):\n{fim}")
 99    with open(ficheiro, encoding="utf-8") as f:
100        return json.load(f)

103def linha(nome, resumo):
104    """Uma linha da tabela, a partir do resumo de uma execução.
105
106    Args:
107        nome (str): Nome do experimento.
108        resumo (dict): O `resumo.json` da execução.
109
110    Returns:
111        dict: Os valores de `CAMPOS`.
112    """
113    deteccoes = [v for v in resumo["tempo_deteccao_s"].values() if v is not None]
114    propostas = resumo["propostas"]
115    return {
116        "experimento": nome,
117        "cenario": resumo["cenario"],
118        "mensagens": resumo["mensagens"]["total"],
119        "alertas": resumo["alertas"],
120        "deteccao_media_s": (round(sum(deteccoes) / len(deteccoes), 1)
121                             if deteccoes else ""),
122        "votacoes_aprovadas": propostas["aprovadas"],
123        "votacoes_rejeitadas": propostas["rejeitadas_votacao"],
124        "descargas": propostas["descargas"],
125        "recursos_usados": sum(resumo["recursos"]["total"].values()),
126        "transacoes": resumo["transacoes"]["total"],
127        "nivel_max_A": round(resumo["nivel_maximo"]["A"], 1),
128        "nivel_max_B": round(resumo["nivel_maximo"]["B"], 1),
129        "nivel_max_C": round(resumo["nivel_maximo"]["C"], 1),
130        "ciclos_em_risco_elevado": sum(resumo["ciclos_em_risco_elevado"].values()),
131    }

134def juntar(existentes, nova):
135    """Acrescenta uma linha à tabela, substituindo a do mesmo experimento e cenário.
136
137    Args:
138        existentes (list): Linhas já na tabela.
139        nova (dict): A linha a acrescentar.
140
141    Returns:
142        list: As linhas, por cenário, com "base" primeiro em cada cenário.
143    """
144    chave = (nova["experimento"], nova["cenario"])
145    linhas = [l for l in existentes if (l["experimento"], l["cenario"]) != chave]
146    linhas.append(nova)
147    return sorted(linhas, key=lambda l: (l["cenario"], l["experimento"] != "base",
148                                         l["experimento"]))

151def ler_tabela():
152    """Lê a tabela gravada, ou uma lista vazia se ainda não existe.
153
154    Returns:
155        list: Uma linha (dict) por execução já gravada.
156    """
157    if not os.path.isfile(TABELA):
158        return []
159    with open(TABELA, newline="", encoding="utf-8") as f:
160        return list(csv.DictReader(f))

163def gravar_tabela(linhas):
164    """Escreve a tabela em resultados/experimentos/comparacao.csv.
165
166    Args:
167        linhas (list): Uma linha (dict) por execução, com os `CAMPOS`.
168    """
169    os.makedirs(PASTA, exist_ok=True)
170    with open(TABELA, "w", newline="", encoding="utf-8") as f:
171        escritor = csv.DictWriter(f, fieldnames=CAMPOS)
172        escritor.writeheader()
173        escritor.writerows(linhas)

176def mostrar(linhas):
177    """Escreve a tabela no terminal, alinhada.
178
179    Args:
180        linhas (list): Uma linha (dict) por execução, com os `CAMPOS`.
181    """
182    cabecalho = ["experimento", "cenario", "msgs", "alertas", "deteção s", "votos ok",
183                 "votos não", "descargas", "recursos", "transações", "A", "B", "C",
184                 "ciclos risco"]
185    tabela = [cabecalho] + [[str(l[c]) for c in CAMPOS] for l in linhas]
186    larguras = [max(len(linha_[i]) for linha_ in tabela) for i in range(len(cabecalho))]
187    for linha_ in tabela:
188        print("  ".join(v.ljust(w) for v, w in zip(linha_, larguras)))

191def main(argv):
192    """Escolhe os grupos, corre-os e grava a tabela depois de cada execução.
193
194    Args:
195        argv (list): Argumentos da linha de comandos: nomes de grupos e a
196            opção `--lista` (só mostra o plano).
197    """
198    grupos = [a for a in argv if not a.startswith("-")] or list(GRUPOS)
199    desconhecidos = [g for g in grupos if g not in GRUPOS]
200    if desconhecidos:
201        sys.exit(f"Grupo desconhecido: {', '.join(desconhecidos)}. "
202                 f"Opções: {', '.join(GRUPOS)}")
203    plano = [(nome, cenario, params) for g in grupos for nome, cenario, params in GRUPOS[g]]
204
205    if "--lista" in argv:
206        for nome, cenario, params in plano:
207            alteracao = "(sem alterações)" if not params else ", ".join(params)
208            print(f"{cenario:20s} {nome:18s} {alteracao}")
209        print(f"{len(plano)} execuções, cerca de {len(plano) * 1.5:.0f} minutos")
210        return
211
212    linhas = ler_tabela()
213    for i, (nome, cenario, params) in enumerate(plano, 1):
214        print(f"[{i}/{len(plano)}] {nome} em {cenario}...", flush=True)
215        try:
216            linhas = juntar(linhas, linha(nome, correr(nome, cenario, params)))
217        except (RuntimeError, subprocess.TimeoutExpired) as erro:
218            print(f"  falhou: {erro}", flush=True)
219            continue
220        gravar_tabela(linhas)           # guarda já, para não perder o que correu
221    print()
222    mostrar(linhas)
223    print(f"\nTabela em {os.path.relpath(TABELA, RAIZ)}")
